func TestValidateKey_InvalidAPIKey(t *testing.T) {
    // Arrange
    ctx := context.Background()
    apiRequest := &models.APIKeyReq{
        APIKey: "invalid-api-key",
    }
    mockUserDB := new(platformMocks.UserDB)
    mockCompanyDB := new(platformMocks.CompanyDB)
    mockLogger := zap.NewNop()

    mockUserDB.On("FindByKey", ctx, "invalid-api-key").Return(nil, errors.New("user not found"))

    authSvc := New(mockUserDB, mockCompanyDB, mockLogger)

    // Act
    res, err := authSvc.ValidateKey(ctx, apiRequest)

    // Assert
    require.Error(t, err)
    assert.NotNil(t, res)
    assert.False(t, res.IsValid)
    assert.Equal(t, models.ErrAuth, err)

    mockUserDB.AssertExpectations(t)
    mockCompanyDB.AssertExpectations(t)
}

func TestValidateKey_ValidUser(t *testing.T) {
    // Arrange
    ctx := context.Background()
    apiRequest := &models.APIKeyReq{
        APIKey: "valid-api-key",
    }
    mockUserDB := new(platformMocks.UserDB)
    mockCompanyDB := new(platformMocks.CompanyDB)
    mockLogger := zap.NewNop()

    user := &models.User{
        Name:   "Test User",
        Email:  "test@example.com",
        Status: models.UserStatusActive,
        Role:   models.UserRoleCICD,
        Cid:    "test-company-id",
    }

    mockUserDB.On("FindByKey", ctx, "valid-api-key").Return(user, nil)
    mockUserDB.On("MapKeywithID", ctx, "valid-api-key", "").Return(true, nil)
    mockCompanyDB.On("GetCompanyByCid", ctx, "test-company-id").Return(&models.Company{}, nil)

    authSvc := New(mockUserDB, mockCompanyDB, mockLogger)

    // Act

    // Assert

    assert.NotNil(t, res)
    assert.True(t, res.IsValid)
    assert.NotEmpty(t, res.JwtToken)

    mockUserDB.AssertExpectations(t)
    mockCompanyDB.AssertExpectations(t)
}
