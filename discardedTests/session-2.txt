func TestFindLastCreated_Success(t *testing.T) {
    ctx := context.Background()
    appID := "app-id"

    mockTestSetDB := new(mockPlatform.TestSetDB)
    mockLogger := zap.NewNop()

    expectedTestSet := models.TestSet{
        ID:    "last-created-id",
        Name:  "Last Created Test Set",
        AppID: appID,
    }

    mockTestSetDB.On("FindLastCreated", ctx, appID).
        Return(expectedTestSet, nil)

    service := &testSetService{
        testSetDb: mockTestSetDB,
        logger:    mockLogger,
    }

    testSet, err := service.FindLastCreated(ctx, appID)

    require.NoError(t, err)
    assert.Equal(t, expectedTestSet, testSet)

    mockTestSetDB.AssertExpectations(t)
}

func TestFindLastCreated_Error(t *testing.T) {
    ctx := context.Background()
    appID := "app-id"

    mockTestSetDB := new(mockPlatform.TestSetDB)
    mockLogger := zap.NewNop()

    mockTestSetDB.On("FindLastCreated", ctx, appID).
        Return(models.TestSet{}, errors.New("database error"))

    service := &testSetService{
        testSetDb: mockTestSetDB,
        logger:    mockLogger,
    }

    testSet, err := service.FindLastCreated(ctx, appID)

    require.Error(t, err)
    assert.Equal(t, models.TestSet{}, testSet)

    mockTestSetDB.AssertExpectations(t)
}
