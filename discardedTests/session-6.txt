        Cid:  "test-cid",
        Role: string(models.UserRoleAdmin),
    })

    reader := io.NopCloser(strings.NewReader("mock data"))
    fileName := "mock-file.json"
    appName := "test-app"

    mockBlobDB := new(platform.BlobDB)
    mockMockRegistryDB := new(platform.MockRegistryDB)
    logger := zap.NewNop()

    mockBlobDB.On("Upload", ctx, reader, "test-cid", filepath.Join(appName, fileName)).
        Return(false, nil)

    mockMockRegistryDB.On("Insert", ctx, models.MockRegistry{
        Mock:      fileName,
        App:       appName,
        Company:   "test-cid",
        MockType:  models.PRIVATE_MOCK,
        CreatedAt: mock.AnythingOfType("int"),
    }).Return(true, nil)

    storage := New(mockBlobDB, mockMockRegistryDB, logger)

    err := storage.MockUpload(ctx, reader, fileName, appName)

    require.NoError(t, err)
    mockBlobDB.AssertExpectations(t)
    mockMockRegistryDB.AssertExpectations(t)
}
