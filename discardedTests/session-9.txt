func TestCreateUserSuccess(t *testing.T) {
    ctx := context.WithValue(context.Background(), models.CustomClaimsKey, models.Claims{
        Cid:   "test-cid",
        Email: "admin@example.com",
    })

    adminUser := &models.User{
        Email: "admin@example.com",
        Cid:   "test-cid",
    }

    newUser := models.User{
        Name:  "Test User",
        Email: "test@example.com",
        Role:  models.UserRoleUser,
        Cid:   "test-cid",
    }

    mockUserDB := new(platformMocks.UserDB)
    mockCompanyDB := new(platformMocks.CompanyDB)
    mockLogger := zap.NewNop()

    mockUserDB.On("FindByEmail", ctx, "admin@example.com").Return(adminUser, nil)
    mockCompanyDB.On("CreateCompany", ctx, mock.Anything).Return(true, nil)
    mockUserDB.On("Create", ctx, mock.Anything, false).Return(true, nil)

    service := &userService{
        udb:    mockUserDB,
        cdb:    mockCompanyDB,
        logger: mockLogger,
    }

    created, err := service.Create(ctx, newUser)

    require.NoError(t, err)
    require.True(t, created)

    mockUserDB.AssertExpectations(t)
    mockCompanyDB.AssertExpectations(t)
}
